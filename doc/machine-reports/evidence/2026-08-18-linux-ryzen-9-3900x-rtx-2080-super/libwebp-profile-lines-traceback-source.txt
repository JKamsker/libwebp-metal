                                   169. // Empirical value to avoid high memory consumption but good for performance.
                                   170. #define COST_CACHE_INTERVAL_SIZE_MAX 500

                                   172. // To perform backward reference every pixel at index 'index' is considered and
                                   173. // the cost for the MAX_LENGTH following pixels computed. Those following pixels
                                   174. // at index 'index' + k (k from 0 to MAX_LENGTH) have a cost of:
                                   175. //     cost = distance cost at index + GetLengthCost(cost_model, k)
                                   176. // and the minimum value is kept. GetLengthCost(cost_model, k) is cached in an
                                   177. // array of size MAX_LENGTH.
                                   178. // Instead of performing MAX_LENGTH comparisons per pixel, we keep track of the
                                   179. // minimal values using intervals of constant cost.
                                   180. // An interval is defined by the 'index' of the pixel that generated it and
                                   181. // is only useful in a range of indices from 'start' to 'end' (exclusive), i.e.
                                   182. // it contains the minimum value for pixels between start and end.
                                   183. // Intervals are stored in a linked list and ordered by 'start'. When a new
                                   184. // interval has a better value, old intervals are split or removed. There are
                                   185. // therefore no overlapping intervals.
                                   186. typedef struct CostInterval CostInterval;
                                   187. struct CostInterval {
                                   188.   int64_t cost;
                                   189.   int start;
                                   190.   int end;
                                   191.   int index;
                                   192.   CostInterval* previous;
                                   193.   CostInterval* next;
                                   194. };
